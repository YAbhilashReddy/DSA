#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;cin>>t;
    while(t--){
        int n,d;cin>>n>>d;
        string s;cin>>s;
        int idx = n;
        for(int i=0 ; i<n ; i++){
            if(s[i] - '0' < d) {idx = i ; break;}
        }
        s.insert(idx , 1 , char('0' + d));
        cout << s << '\n';
    }
  return 0; 
}
