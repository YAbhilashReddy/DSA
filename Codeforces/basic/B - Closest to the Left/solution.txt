#include<bits/stdc++.h>
using namespace std;

void slove(){
    int n,k;cin>>n>>k;
    vector<int> arr(n);
    for(int i=0 ; i<n ; i++) cin>>arr[i];
    while(k--){
        int x;cin>>x;
        auto it = upper_bound(arr.begin() , arr.end() , x);
        cout << it - arr.begin() << '\n';
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    int t = 1;

    while(t--) {
        slove();
    }
    
  return 0;
}
