#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int n;cin>>n;
    int one = 0;
    vector<int> arr;
    for(int i = 0 ; i < n ; i++){
        int m;cin>>m;
        if(m == 1) one++ , arr.push_back(-1);
        else arr.push_back(1);
    }
    int sum = 0 , maxSum = INT_MIN;
    for(int i:arr) sum = max(sum + i , i) , maxSum = max(maxSum , sum);
    cout << maxSum + one << endl;
  return 0;
}
