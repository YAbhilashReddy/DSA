#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    bool r = true;
    for(int i=0 ; i<8 ; i++){
        string s;cin>>s;
        for(int i=1 ; i<8 ; i++){
            if(s[i] == s[i-1]) {r = false; break;}
        }
        if(!r)break;
    }
    cout << (r ? "YES" : "NO") << '\n';
  return 0;
}
