#include<bits/stdc++.h>
using namespace std;

void slove(){
    int n;
    long long L, a;
    cin >> n >> L >> a;
    long long ans = 0, prev = 0;
    for (int i = 0; i < n; i++) {
        long long t, l;cin >> t >> l;
        ans += (t - prev) / a;
        prev = t + l;
    }
    ans += (L - prev) / a;
    cout << ans << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    int t = 1;
    while(t--) {
        slove();
    }
    
  return 0;
}
