#include <iostream>
#include <string>
using namespace std;

int main(){
  int t;
  cin>>t;
  for(int i=0;i<t;i++){
    int n;
    string str;
    cin>>n;
    cin>>str;
    int three = 0;
    int count = 0;
    int isEmpty = 0;
    int isInfinite = 0;
    for(int i=0;i<str.length();i++){
      if(str[i]=='.'){
        isEmpty = 1;
        count++;
      }
      if(str[i]=='.'){
        three++;
        if(three == 3){
          isInfinite = 1;
        }
      }else{
        three = 0;
      }
    }
    if(!isEmpty){
      cout<<0<<endl;
    }else if(isInfinite){
      cout<<2<<endl;
    }else{
      cout<<count<<endl;
    }

  }
  return 0;
}
