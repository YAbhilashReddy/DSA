#include<bits/stdc++.h>
using namespace std;
#define int long long
bool check(int n){
    int temp = n;
    while(n){
        int val = n%10;
        if(val!=0 && temp%val!=0)return false;
        n/=10;
    }return true;
}
signed main() {
  int t; cin>>t;
  while(t--){
      int n; cin>>n;
      while(!check(n))n++;
      cout<<n<<"\n";
  }
  return 0;
}
