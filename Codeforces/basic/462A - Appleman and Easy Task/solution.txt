#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n;cin>>n;
    vector<string> arr(n);
    for(int i = 0 ; i < n ; i++) cin>>arr[i];
    for(int i = 0 ; i < n ; i++) {
        for(int j = 0 ; j < n ; j++) {
            int count = 0;
            
            //UP :
            if(i > 0 && arr[i-1][j] == 'o') count++;
            
            //DOWN :
            if(i < n-1 && arr[i+1][j] == 'o') count++;
            
            //LEFT :
            if(j > 0 && arr[i][j-1] == 'o') count++;
            
            //RIGHT :
            if(j < n-1 && arr[i][j+1] == 'o') count++;
            
            if(count % 2 != 0) {cout << "NO\n" ; return ;}
        }
    }
    cout << "YES\n";
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
