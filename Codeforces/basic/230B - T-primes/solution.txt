#include<bits/stdc++.h>
using namespace std;

bool prime(int N){
    if(N < 2) return false;
    for(int i=2 ; i*i<=N ; i++){
        if(N % i == 0) return false;
    }
    return true;
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;cin>>t;
    while(t--){
        long long n;cin>>n;
        int x = sqrt(n);
        cout << ((long long)x*x == n && prime(x) ? "YES" : "NO") << '\n';
    }
  return 0;
}
