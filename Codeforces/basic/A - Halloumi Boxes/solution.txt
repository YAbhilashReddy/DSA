#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int t;cin>>t;
    while(t--){
        int n,k;cin>>n>>k;
        vector<int> arr(n);
        for(int i=0 ; i<n ; i++) cin>>arr[i];
        string ans = "YES";
        if(k == 1){
            if(!is_sorted(arr.begin() , arr.end())) ans = "NO";
        }
        cout << ans << endl;
    }
  return 0;
}
