#include <bits/stdc++.h>
using namespace std;

#define int long long
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(NULL);

    int t;
    cin >> t;

    while (t--)
    {
        int n;
        cin >> n;
        vector<int> x(n), y(n);
        for (int i = 0; i < n; i++)
            cin >> x[i];
        for (int i = 0; i < n; i++)
            cin >> y[i];

        int d = 0;
        for (int i = n - 1; i >= 0; i--)
        {
            int  diff = x[i] - y[i];
            d = max(diff + d, -d);
        }

        if (d > 0)
        {
            cout << "Sunny\n";
        }
        else if (d < 0)
        {
            cout << "Aubrey\n";
        }
        else
        {
            cout << "Draw\n";
        }
    }
    return 0;
}
