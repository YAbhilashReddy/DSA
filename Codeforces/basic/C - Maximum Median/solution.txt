#include<bits/stdc++.h>
using namespace std;

bool checkFun(vector<int> &arr , int mid , int rem){
    int n = arr.size() , target = arr[n/2] + mid;
    long long count = 0;
    for(int i = (n/2)+1 ; i<n ; i++){
        if(target > arr[i]) count += target - arr[i];
    }
    return count <= rem;
}
    
void slove(){
    int n,k;cin>>n>>k;
    vector<int> arr(n);
    for(int i=0 ; i<n ; i++) cin>>arr[i];
    sort(arr.begin() , arr.end());
    int low = 0 , high = k , ans = arr[n/2];
    while(low <= high){
        int mid = low + (high - low) / 2;
        if(checkFun(arr , mid , k-mid)){
            ans = arr[n / 2] + mid;
            low = mid + 1;
        } else high = mid - 1;
    }
    cout << ans << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    slove();
    
  return 0;
}
