#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    string a,b;cin>>a>>b;
    if(a.length() != b.length()) {cout << "NO" ; return 0;}
    int c = 0;
    for(int i=0 ; i<a.length() ; i++) c += (a[i] != b[i]);
    string ans;
    if(c == 2){
        sort(a.begin() , a.end());
        sort(b.begin() , b.end());
        ans = (a == b ? "YES" : "NO");
    }
    else if(c == 0){
        bool r = false;
        vector<int> arr(26,0);
        for(char c:a) arr[c - 'a']++;
        for(int i:arr){
            if(i == 2) r = true;
        }
        ans = (r ? "YES" : "NO");
    } else ans = "NO";
    cout << ans << '\n';
  return 0;
}
