#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;cin>>n;
    unordered_map<int,int> freq;
    int maxi = INT_MIN;
    for(int i=0 ; i<n ; i++){
        int m;cin>>m;
        freq[m]++;
        maxi = max(maxi , freq[m]);
    }
    cout << maxi << ' ' << freq.size() << '\n';
  return 0;
}
