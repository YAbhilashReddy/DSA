#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n;cin>>n;
    int sum = 0 , first , last ;
    for(int i = 0 ; i < n ; i++) {
        int x;cin>>x;
        if(i == 0) first = x;
        if(i == n-1) last = x;
        sum += x;
    }
    if(sum == n) cout << -1 << '\n';
    else if(sum == n - 2 + first + last) cout << first + last - 1 << '\n';
    else cout << sum - 1 << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    int t = 1;
    cin>>t;
    
    while(t--) {
        solve();
    }
    
  return 0;
}
