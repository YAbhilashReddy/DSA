#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;cin>>t;
    while(t--){
        int n,k;cin>>n>>k;
        vector<pair<int,int>> arr;
        for(int i=0 ; i<n ; i++){
            int m;cin>>m;
            int rem = (m - 1) % k + 1;
            arr.push_back({rem , i+1});
        }
        sort(arr.begin() , arr.end(), [](pair<int,int> &a , pair<int,int> &b){
            if(a.first == b.first) return a.second < b.second;
            return a.first > b.first;
        });
        for(auto [k,v]:arr) cout << v << ' ';
        cout << endl;
    }
  return 0;
}
