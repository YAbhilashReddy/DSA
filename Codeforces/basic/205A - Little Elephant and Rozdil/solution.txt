#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n;cin>>n;
    unordered_map<int,int> freq;
    int idx = 0 , mini = INT_MAX;
    for(int i = 0 ; i < n ; i++) {
        int x;cin>>x;
        freq[x]++;
        if(x < mini) mini = x , idx = i + 1;
    }
    if(freq[mini] == 1) cout << idx << '\n';
    else cout << "Still Rozdil\n";
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
