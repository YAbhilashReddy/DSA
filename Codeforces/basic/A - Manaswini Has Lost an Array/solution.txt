#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,l,r;cin>>n>>l>>r;
    long long mini = (n - l + 1) , maxi = 0;
    for(int i=1 ; i<l ; i++) mini += (long long)pow(2,i);
    for(int i=0 ; i<r ; i++) maxi += (long long)pow(2,i);
    maxi += (n-r) * (long long)pow(2,r-1);
    cout << mini << ' ' << maxi << '\n';
  return 0;
}
