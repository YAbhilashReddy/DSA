#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int n;cin>>n;
    string s;cin>>s;
    int SF = 0 , FS = 0;
    for(int i = 0 ; i < n-1 ; i++){
        if(s[i] == 'S' && s[i+1] == 'F') SF++;
        else if(s[i] == 'F' && s[i+1] == 'S') FS++;
    }
    cout << (SF > FS ? "YES" : "NO") << endl;
  return 0;
}
