#include<bits/stdc++.h>
using namespace std;

long long mod = 1e9 + 7;

void solve(){
    long long n;cin>>n;
    // Brute force is too slow for large constraints,
    // so we use a maths formula to calculate the answer in O(1) constant per test case.
    // Formula: 2022 * n * (n + 1) * (4 * n - 1) / 6
    // (1^2) + (1*2) + (2^2) + (2*3) .....
    
    long long ans = 337; //this number comes from formula : 2022 / 6
    
    ans = ans * (n % mod) % mod;
    ans = ans * ((n + 1) % mod ) % mod;
    ans = ans * ((4 * n - 1) % mod) % mod;
    
    cout << ans << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    int t = 1;
    cin>>t;
    
    while(t--) {
        solve();
    }
    
  return 0;
}
