#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int t;cin>>t;
    while(t--){
        int n,q;cin>>n>>q;
        vector<int> arr(n) , prefix(n);
        for(int i=0 ; i<n ; i++) cin>>arr[i];
        sort(arr.rbegin() , arr.rend());
        for(int i=0 ; i<n ; i++) prefix[i] = (i ? prefix[i-1] : 0) + arr[i];
        int max = prefix[n-1];
        for(int i=0 ; i<q ; i++){
            int lim;cin>>lim;
            int ans = 0;
            if(lim > max) ans = -1;
            else ans =( lower_bound(prefix.begin() , prefix.end() , lim) - prefix.begin()) + 1;
            cout << ans << endl;
        }
    }
  return 0;
}
