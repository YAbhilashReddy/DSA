#include<bits/stdc++.h>
using namespace std;

void solve(){
    int day = 0;
    int n;cin>>n;
    while(n--) {
        int s,d;cin>>s>>d;
        if(s > day) day = s;
        else {
            while(day >= s) s += d;
            day = s;
        }
    }
    cout << day << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
