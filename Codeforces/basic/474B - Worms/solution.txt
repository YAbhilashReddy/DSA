#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;cin>>n;
    vector<pair<long long , long long>> arr(n+1);
    for(int i=1 ; i<=n ; i++){
        int a;cin>>a;
        if(i == 1) arr[i] = {1,a};
        else arr[i] = {arr[i-1].second+1 , arr[i-1].second+a};
    }
    int m;cin>>m;
    for(int i=0 ; i<m ; i++){
        long long q;cin>>q;
        int low = 1 , high = n;
        while(low <= high){
            int mid = low + (high - low) / 2;
            long long f = arr[mid].first , s = arr[mid].second;
            if(q >= f && q <= s) {cout << mid << "\n";break;}
            else if(q < f) high = mid-1;
            else low = mid+1;
        }
    }
    return 0;
}
