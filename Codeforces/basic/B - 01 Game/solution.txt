#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int t;cin>>t;
    while(t--){
        string s;cin>>s;
        string S = "";
        int count = 0;
        for(char c:s){
            if(S.empty()) S.push_back(c);
            else if(S.back() != c) S.pop_back() , count++;
            else S.push_back(c);
        }
        cout << (count % 2 == 1 ? "DA" : "NET") << '\n';
    }
  return 0;
}
