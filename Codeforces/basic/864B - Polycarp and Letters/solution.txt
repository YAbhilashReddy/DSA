#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;cin>>n;
    string s;cin>>s;
    int maxi = INT_MIN;
    unordered_set<char> st;
    for(char c:s){
        if(c >= 'a' && c  <= 'z') {
            st.insert(c);
            int N = st.size();
            maxi = max(maxi , N);
        }
        else {
            int N = st.size();
            maxi = max(maxi , N);
            st.clear();
        }
    }
    int N = st.size();
    maxi = max(maxi , N);
    cout << maxi << endl;
  return 0;
}
