#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int t;cin>>t;
    while(t--){
        string s;cin>>s;
        unordered_map<char,int> freq;
        for(char c:s) freq[c]++;
        int one = 0 , two = 0;
        for(auto[k,v]:freq){
            if(v == 1) one++;
            else if(v >= 2) two++;
        }
        two += (one / 2);
        cout << two << endl;
    }
    return 0;
}
