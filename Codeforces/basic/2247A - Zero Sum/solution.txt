#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;cin>>t;
    while(t--){
        int n;cin>>n;
        int total = 0;
        for(int i=0 ; i<n ; i++) {
            int x;cin>>x;
            total += x;
        }
        if(n % 2) cout << "NO\n";
        else cout << (abs(total) % 4 == 0 ? "YES" : "NO") << '\n';
        // key observation : that every operation flips two adjacent elements.
        // there will be only 3 way to get sum = 0, => -4 , 0 , 4 
    }
  return 0;
}
