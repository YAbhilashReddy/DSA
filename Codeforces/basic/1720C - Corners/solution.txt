void solve() {
   int n, m;
   cin >> n >> m;
   vs grid(n);
   int total_ones = 0;

   for (int i = 0; i < n; ++i)
   {
       cin >> grid[i];
       for (char c : grid[i])
       {
           if(c == '1') total_ones++;
       }
   }
   if(total_ones == 0){
    cout << 0 << "\n";
    return;
   }
   int minL = 3;
   for (int i = 0; i < n - 1; i++)
   {
       for (int j = 0; j < m - 1; j++)
       {
           int cnt = (grid[i][j] - '0') + (grid[i][j + 1] - '0') + (grid[i + 1][j] - '0') + (grid[i + 1][j + 1] - '0');
           int ones = (cnt == 4 ? 3 : max(0, cnt - 1));
           minL = min(minL, ones);
       }
   }
   if (minL <= 1) {
        cout << total_ones << "\n";
    } else if (minL == 2) {
        cout << total_ones - 1 << "\n";
    } else {
        cout << total_ones - 2 << "\n";
    }
}

int main() {
    fast_io();

    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
