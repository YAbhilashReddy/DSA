#include <iostream>
#define ll long long 
using namespace std;
constexpr int N=1e5+2;
int T;
ll n;
int main() {
	ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
	cin>>T;
	while(T--) {
		cin>>n;
		if(n==1) cout<<"1\n";
		else if(n==2) cout<<"-1\n";
		else {
			cout<<"1 2 ";
			for(ll i=0;i<=n-3;i++) cout<<(1LL<<i)*3<<" \n"[i==n-3];
		}
	}
	return 0;
}
