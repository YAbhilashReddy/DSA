#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;cin>>t;
    while(t--){
        int n;cin>>n;
        string s;cin>>s;
        bool ans = s[0] == '1' || s[n-1] == '1';
        for(int i=1 ; i<n ; i++){
            if(s[i] == '1' && s[i-1] == '1') {ans = true; break;}
        }
        cout << (ans ? "YES" : "NO") << '\n';
    }
  return 0;
}
