#include<bits/stdc++.h>
using namespace std;

void slove(){
    int n;cin>>n;
    vector<int> arr(n);
    for(int i=0 ; i<n ; i++) cin>>arr[i];
    sort(arr.begin() , arr.end());
    int k;cin>>k;
    while(k--){
        int l,r;cin>>l>>r;
        auto left = lower_bound(arr.begin() , arr.end() , l);
        auto right = upper_bound(arr.begin() , arr.end() , r);
        cout << right - left << ' ';
    }
    cout << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
        
    int t = 1;

    while(t--) {
        slove();
    }
    
  return 0;
}
