int pw(int a , int b){
    if (b==0) return 1;
    int p = pw(a,b/2);
    if (b&1) return ((p*p)%M * a)%M;
    return (p*p)%M;
}

void solve(){
    cin>>n;
    
    vec<int> x,y;

    for (int i=0;i<n;i++){
        cin>>u>>v;
        x.pb(u);
        y.pb(v);
    }

    if (n&1){
        cout<<1<<endl;
    }else{
        sort(all(x));sort(all(y));

        int ans=(x[n/2]-x[n/2 - 1]+1)*(y[n/2] - y[n/2 - 1]+1);
        cout<<ans<<endl;
    }
}
