#include<bits/stdc++.h>
using namespace std;

//finding if "AB" & "BA" are exist in the given string or not :
//if "YES" we are finding the positon / index :
//checking if the positions are overlapping are not with there index position

void solve(){
    string s;cin>>s;
    int ab = s.find("AB") , ba = s.find("BA");
    string ans = "NO";
    if(ab != -1 && s.find("BA" , ab + 2) != -1) ans = "YES";
    else if(ba != -1 && s.find("AB" , ba + 2) != -1) ans = "YES";
    cout << ans << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
