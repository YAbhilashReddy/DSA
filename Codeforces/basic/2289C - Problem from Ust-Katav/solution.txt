int n;
stack<int>sta;
int R[MAXN + 1];
int L[MAXN + 1];
int a[MAXN + 1];

void build() {
    for (int i = 1;i <= n;i++) {
        while (sta.size() && a[i] < a[sta.top()]) {
            int t = sta.top();
            R[t] = i;
            sta.pop();
        }

        if (sta.size()) {
            if (a[i] == a[sta.top()]) {
                L[i] = L[sta.top()];
            }
            else {
                L[i] = sta.top();
            }
        }
        else {
            L[i] = -1;
        }
        sta.push(i);
    }
    while (sta.size()) {
        int t = sta.top();
        R[t] = -1;
        sta.pop();
    }
}


void solve() {

    cin >> n;
    for (int i = 1;i <= n;i++) {
        cin >> a[i];
    }
    build();
    ll ans = n;
    for (int i = 1;i <= n;i++) {
        int l = L[i] != -1 ? L[i] : 0;
        int r = R[i] != -1 ? R[i] : n + 1;
        if (l == i - 1 && r == i + 1) {
            continue;
        }
        if (l == i - 1) {
            ans = ans + (r - i - 1);
        }
        else if (r == i + 1) {
            ans = ans + (i - l - 1);
        }
        else {
            ans = ans + 1;
        }

    }
    cout << ans << '\n';
}
