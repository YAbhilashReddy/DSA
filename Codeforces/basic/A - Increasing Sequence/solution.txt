#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,d;cin>>n>>d;
    long long c = 0;
    vector<long long> arr(n);
    for(int i=0 ; i<n ; i++) cin>>arr[i];
    for(int i=1 ; i<n ; i++){
        if(arr[i] <= arr[i-1]){
            long long diff = arr[i-1] - arr[i] + 1; //we want strictly greater, not equal
            long long k = (diff + d - 1) / d; // we want ceil value...
            arr[i] += k * d;
            c += k;
        }
    }
    cout << c << endl;
  return 0;
}
