#include<bits/stdc++.h>
using namespace std;
 
int main() {
    string s;cin>>s;
    vector<int> arr , sorted;
    int sum = 0;
    for(char c:s){
        if(c != '+') {
            int n = c - '0';
            sum += n;
            arr.push_back(n) , sorted.push_back(n);
        }
    }
    sort(sorted.begin() , sorted.end());
    if(arr != sorted) {
        for(int i=0 ; i<sorted.size() ; i++){
            cout << sorted[i];
            if(i+1 != sorted.size()) cout << '+';
        }
        cout << endl;
    } else cout << s << endl;
  return 0;
}
