#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;cin>>t;
    vector<pair<int,int>> arr(t);
    for(int i=0 ; i<t ; i++) cin>>arr[i].first >> arr[i].second;
    sort(arr.begin() , arr.end());
    for(int i=1 ; i<t ; i++){
        if(arr[i].second < arr[i-1].second) {
            cout << "Happy Alex\n";
            return 0;
        }
    }
    cout << "Poor Alex\n";
  return 0;
}
