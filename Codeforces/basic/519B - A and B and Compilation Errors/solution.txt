#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n;cin>>n;
    long long first = 0 , second = 0 , third = 0;
    for(int i = 0 ; i < n ; i++) { int x;cin>>x; first += x; }
    for(int i = 0 ; i < n-1 ; i++) { int x;cin>>x; second += x; }
    for(int i = 0 ; i < n - 2 ; i++) { int x;cin>>x; third += x; }
    cout << first - second << '\n';
    cout << second - third << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
