#include<bits/stdc++.h>
using namespace std;

void slove(){
    int n,k;cin>>n>>k;
    vector<int> isprime(n+1 , true);
    isprime[0] = false , isprime[1] = false;
    for(int i=2 ; i*i<=n ; i++){
        if(isprime[i]){
            for(int j=i*i ; j<=n ; j += i) isprime[j] = false;
        }
    }
    vector<int> primes;
    for(int i=2 ; i<=n ; i++) {
        if(isprime[i]) primes.push_back(i);
    }
    int c = 0;
    for(int i=1 ; i<primes.size() ; i++){
        int sum = primes[i] + primes[i-1] + 1;
        c += (sum <= n && isprime[sum]);
    }
    cout << (c >= k ? "YES" : "NO") << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    slove();
    
  return 0;
}
