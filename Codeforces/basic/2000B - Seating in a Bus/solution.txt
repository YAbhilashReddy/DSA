#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<bool> occupied(n + 2, false);
    // First passenger can sit anywhere
    occupied[a[0]] = true;
    for (int i = 1; i < n; i++) {
        int x = a[i];
        bool left = occupied[x - 1];
        bool right = occupied[x + 1];
        if (!left && !right) {
            cout << "NO\n";
            return;
        }
        occupied[x] = true;
    }
    cout << "YES\n";
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
