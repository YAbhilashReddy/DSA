#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int t;cin>>t;
    while(t--){
        int n;cin>>n;
        vector<int> arr(n);
        int z = 0 , o = 0;
        for(int i=0 ; i<n ; i++) cin>>arr[i] , z += (arr[i] == 0) , o += (arr[i] == 1);
        int res;
        if(z >= 2){
            if(arr[0] == 0 && arr[n - 1] == 0) res = 0;
            else if((arr[0] == 0 && arr[n-1] != 0) || (arr[0] != 0 && arr[n-1] == 0)) res = 1;
            else if(arr[0] != 0 && arr[n-1] != 0) res = 2;
        } else res = -1;
        cout << res << '\n';
    }
  return 0;
}
