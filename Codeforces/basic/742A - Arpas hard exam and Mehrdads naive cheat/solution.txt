#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n;cin>>n;
    if(!n) cout << 1 << '\n';
    else {
        int x = n % 4 , ans = 8;
        if(x == 2) ans = 4;
        else if(x == 3) ans = 2;
        else if(!x) ans = 6;
        cout << ans << '\n';
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
