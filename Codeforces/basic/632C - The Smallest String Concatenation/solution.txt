#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int n;cin>>n;
    vector<string> st;
    for(int i=0 ; i<n ; i++){
        string s;cin>>s;
        st.push_back(s);
    }
    sort(st.begin() , st.end() , [](string a , string b){
        return a+b < b+a;
    });
    string ans = "";
    for(string s:st) ans += s;
    cout << ans << endl;
    return 0;
}
