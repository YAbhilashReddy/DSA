#include<bits/stdc++.h>
using namespace std;

void solve(){
    int a,b;cin>>a>>b;
    if(!a) cout << 0 << '\n';
    else {
        int mini = INT_MAX;
        // At Most 31 operations
        for(int i = 0 ; i < 32 ; i++){
            int x = a , y = b + i;
            if(y == 1) continue;
            int c = i;
            while(x) c++ , x /= y;
            mini = min(mini , c);
        }
        cout << mini << '\n';
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    int t = 1;
    cin>>t;
    
    while(t--) {
        solve();
    }
    
  return 0;
}
