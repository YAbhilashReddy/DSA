#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int n;cin>>n;
    string ans = "NO";
    bool a = true, b = true;
    vector<string> arr;
    for(int i=0 ; i<n ; i++){
        string S;cin>>S;
        string f = S.substr(0,2) , s = S.substr(3,2);
        if(f == "OO" && a) S[0] = '+' , S[1] = '+' , ans = "YES" , a = false , b = false;
        if(s == "OO" && b) S[3] = '+' , S[4] = '+' , ans = "YES" , a = false , b = false;
        arr.push_back(S);
    }
    if(ans == "YES"){
        cout << ans << endl;
        for(auto i:arr) cout << i << endl;
    } else cout << ans << endl;
  return 0;
}
