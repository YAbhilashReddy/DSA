#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    int t;cin>>t;
    while(t--){
        int n;cin>>n;
        vector<int> arr(n);
        for(int i=0 ; i<n ; i++) cin>>arr[i];
        int i = 1 , j = n-2 , A = arr[0] , B = arr[n-1];
        int maxi = (n >= 2 && A == B ? 2 : 0);
        while(i <= j){
            if(A < B) A += arr[i++];
            else B += arr[j--];
            if(A == B) maxi = max(maxi , i + n - j - 1);
        }
        cout << maxi << '\n';
    }
  return 0;
}
