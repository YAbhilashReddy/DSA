#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,k;cin>>n>>k;
    vector<int> arr(n) , prefix(n);
    for(int i=0 ; i<n ; i++) cin>>arr[i] , prefix[i] = (i ? prefix[i-1] : 0) + arr[i];
    int sum = prefix[k-1] , idx = 0;
    for(int i=1 ; i<=n-k ; i++){
        int total = prefix[k-1+i] - prefix[i-1];
        if(total <= sum) sum = total , idx = i;
    }
    cout << idx + 1 << endl;
  return 0;
}
