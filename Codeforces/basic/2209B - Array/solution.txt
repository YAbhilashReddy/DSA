#include <bits/stdc++.h>
using namespace std;

void Solve(){
    int n;cin>>n;
    vector<int>a(n);
    for(auto &x:a)cin>>x;

    vector<int>smaller(n,0),greater(n,0);
    
    for(int i=0;i<n;i++){
        for(int j=i+1;j<n;j++){
            if(a[i]<a[j])smaller[i]++;
            else if(a[i]>a[j])greater[i]++;
        }
    }

    for(int i=0;i<n;i++){
        cout<<max(smaller[i],greater[i])<<" ";
    }
    cout<<'\n';
}

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    int t;cin>>t;
    while(t--)Solve();

    return 0;
}
