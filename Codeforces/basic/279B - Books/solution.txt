#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,t;cin>>n>>t;
    vector<int> arr(n);
    for(int i=0 ; i<n ; i++) cin>>arr[i];
    int left = 0 , sum = 0 , ans = 0;
    for(int right = 0 ; right<n ; right++){
        sum += arr[right];
        while(sum > t) sum -= arr[left] , left++;
        ans = max(ans , right - left + 1);
    }
    cout << ans << '\n';
  return 0;
}
