#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n;cin>>n;
    vector<bool> is_prime(n + 1 , true);
    is_prime[0] = is_prime[1] = false;
    for(int i = 2 ; i * i <= n ; i++){
        if(is_prime[i]) {
            for(int j = i * i ; j <= n ; j += i) is_prime[j] = false;
        }
    }
    vector<int> primes;
    for(int i = 2 ; i <= n ; i++) {
        if(is_prime[i]) primes.push_back(i);
    }
    int c = 0;
    for(int i = 1 ; i <= n ; i++){
        int count = 0;
        for(int j : primes) count += (i % j == 0);
        c += (count == 2);
    }
    cout << c << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
