#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int n;cin>>n;
    vector<long long> arr(n);
    for(int i=0 ; i<n ; i++){
        int m;cin>>m;
        if(i == 0) arr[i] = m;
        else arr[i] = arr[i-1] + m;
    }
    int a,b;cin>>a>>b;
    if(a == 0) cout << (long long)arr[b] << endl; 
    else cout << (long long)(arr[b] - arr[a-1]) << endl;
  return 0;
}
