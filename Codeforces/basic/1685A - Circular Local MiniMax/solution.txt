#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n;cin>>n;
    vector<int> arr(n);
    for(int i = 0 ; i < n ; i++) cin>>arr[i];
    if(n % 2) cout << "NO\n";
    else {
        sort(arr.begin() , arr.end());
        int i = 0 , j = n/2;
        vector<int> ans;
        for(int x = 0 ; x < n ; x++) {
            if(x % 2 == 0) ans.push_back(arr[i++]);
            else ans.push_back(arr[j++]);
        }
        bool r = true;
        for(int i = 1 ; i < n-1 ; i++) {
            bool con1 = ans[i] > ans[i-1] && ans[i] > ans[i+1];
            bool con2 = ans[i] < ans[i-1] && ans[i] < ans[i+1];
            if(con1 || con2) continue;
            else {cout << "NO\n"; return;}
        }
        cout << "YES\n";
        for(int i : ans) cout << i << ' ';
        cout << '\n';
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    int t = 1;
    cin>>t;
    
    while(t--) {
        solve();
    }
    
  return 0;
}
