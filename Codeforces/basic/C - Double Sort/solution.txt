#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {

    /*
    --------------------------------
    THOUGHTS
    --------------------------------

    Observations:
    1. 

    2.

    3.


    */


    int n;
    cin >> n;

    vector<int> a(n), b(n);
    for(int i = 0; i<n; i++){
        cin >> a[i];
    }
    for(int i = 0; i<n; i++){
       cin >> b[i];
    }
    vector<pair<int, int>> ans;
    for(int i = n - 1; i>=1; i--){
        for(int j = 0; j<i; j++){
            if((a[j] > a[j + 1]) || (a[j] == a[j + 1] && b[j] > b[j + 1])){
                swap(a[j], a[j + 1]);
                swap(b[j], b[j + 1]);
                ans.push_back({j, j + 1});
            }
        }
    }
    if(!is_sorted(a.begin(), a.end()) || !is_sorted(b.begin(), b.end())){
        cout << -1 << endl;
        return;
    }
    cout << ans.size() << endl;
    for(auto [l, r] : ans){
        cout << l + 1 << " " << r + 1 << endl;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
