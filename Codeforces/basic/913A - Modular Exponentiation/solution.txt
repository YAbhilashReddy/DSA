#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n,m;cin>>n>>m;
    long long ans = 1 , a = 2 , b = n;
    while(b) {
        if(b % 2) ans *= a;
        a *= a;
        b /= 2;
    }
    cout << (n > 31 ? m : m % ans) << '\n';
}
 
int main() {
    ios_base::sync_with_stdio(false);  // Make cin/cout faster
    cin.tie(NULL);                     // Don't unnecessarily flush cout
    
    solve();
    
  return 0;
}
