#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,m;cin>>n>>m;
    string s,t;cin>>s>>t;
    string ans;
    if(n > m && abs(n - m) > 1) ans = "NO";
    else if(s.find('*') != string::npos){
        int idx = s.find('*') ;
        string A = t.substr(0,idx) , B = s.substr(0,idx);
        n -= (idx + 1);
        string AA = t.substr(m-n) , BB = s.substr(idx+1);
        ans = (A == B && AA == BB ? "YES" : "NO");
    } else ans = (s == t ? "YES" : "NO");
    cout << ans << endl;
  return 0;
}
