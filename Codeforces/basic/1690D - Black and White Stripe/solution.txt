#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;cin>>t;
    while(t--){
        int n,k;cin>>n>>k;
        string s;cin>>s;
        int b = 0 , w = 0 , mini;
        for(int i=0 ; i<k ; i++) b += (s[i] == 'B') , w += (s[i] == 'W');
        mini = w;
        int i = 0;
        for(int j=k ; j<n ; j++){
            if(s[i] == 'B') b--;
            else w--;
            i++;
            b += (s[j] == 'B') , w += (s[j] == 'W');
            mini = min(mini , w);
        }
        cout << mini << '\n';
    }
  return 0;
}
