#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {

    /*
    --------------------------------
    THOUGHTS
    --------------------------------

    Observations:
    1. 

    2.

    3.


    */


    int n, m;
    cin >> n >> m;

    vector<int> a(n);
    for (int &x : a) {
        cin >> x;
    }
    vector<int> b(m);
    for(int i = 0; i<m; i++){
        cin >> b[i];
    }
    vector<int> pref(m + 1, n), suff(m + 1, -1);
    pref[0] = -1;
    int j = 0;
    for(int i = 1; i<=m; i++){
        while(j < n && a[j] < b[i - 1]){
            j++;
        }
        if(j < n){
            pref[i] = j;
            j++;
        }
    }
    if(pref[m] != n){
        cout << 0 << endl;
        return;
    }
    suff[m] = n;
    j = n - 1;
    for(int i = m - 1; i>=0; i--){
        while(j >= 0 && a[j] < b[i]){
            j--;
        }
        if(j >= 0){
            suff[i] = j;
            j--;
        }
    }
    int ans = INT_MAX;
    for(int i = 0; i<m; i++){
        if(pref[i] < suff[i + 1]){
            ans = min(ans, b[i]);
        }
    }
    if(ans == INT_MAX){
        cout << -1 << endl;
        return;
    }
    cout << ans << endl;

}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
