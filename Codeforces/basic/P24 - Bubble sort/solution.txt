#include<bits/stdc++.h>
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;cin>>n;
    vector<int> arr(n);
    for(int i=0 ; i<n ; i++) cin>>arr[i];
    bool swapped;
    for(int i=0 ; i<n-1 ; i++){
        for(int j=0 ; j<n-i-1 ; j++){
            if(arr[j] > arr[j+1]) swap(arr[j] , arr[j+1]);
        }
        for(int x:arr) cout << x << ' ';
        cout << '\n';
    }
  return 0;
}
